cmake_minimum_required(VERSION 3.22)
project(bladerf_x40_radio LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
option(LAB_ENABLE_BLADERF_RF "실제 RF stream 코드 활성화 (기본 OFF)" OFF)
set(OCUDU_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../../upstream/ocudu" CACHE PATH "고정 OCUDU checkout")
get_filename_component(LAB_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
find_package(PkgConfig REQUIRED)
pkg_check_modules(BLADERF REQUIRED IMPORTED_TARGET libbladeRF>=2.6.0)
find_package(Threads REQUIRED)
find_package(GTest REQUIRED)
add_library(ocudu_radio_bladerf_x40 SHARED plugin.cpp stream.cpp usb.cpp)
target_include_directories(ocudu_radio_bladerf_x40 SYSTEM PUBLIC "${OCUDU_SOURCE}/include" "${OCUDU_SOURCE}/external/fmt/include")
target_compile_definitions(ocudu_radio_bladerf_x40 PUBLIC FMT_HEADER_ONLY)
target_compile_options(ocudu_radio_bladerf_x40 PRIVATE -Wall -Wextra -Werror -fno-rtti)
target_link_libraries(ocudu_radio_bladerf_x40 PRIVATE PkgConfig::BLADERF Threads::Threads)
if(LAB_ENABLE_BLADERF_RF)
  target_compile_definitions(ocudu_radio_bladerf_x40 PRIVATE LAB_ENABLE_BLADERF_RF=1)
endif()
add_executable(radio-plugin-tests "${LAB_ROOT}/tests/radio/plugin_test.cpp")
target_link_libraries(radio-plugin-tests PRIVATE ocudu_radio_bladerf_x40 GTest::gtest_main Threads::Threads dl)
target_compile_options(radio-plugin-tests PRIVATE -fno-rtti)
if(LAB_ENABLE_BLADERF_RF)
  target_compile_definitions(radio-plugin-tests PRIVATE LAB_ENABLE_BLADERF_RF=1)
endif()
enable_testing()
add_test(NAME radio-plugin COMMAND radio-plugin-tests)
set_tests_properties(radio-plugin PROPERTIES TIMEOUT 20)
install(TARGETS ocudu_radio_bladerf_x40 LIBRARY DESTINATION lib)
# 실물 RF 코드도 실제 libbladeRF에 링크하되 모든 장치 C API를 test double로 대체한다.
add_executable(radio-usb-tests "${LAB_ROOT}/tests/radio/usb_test.cpp" usb.cpp stream.cpp)
target_include_directories(radio-usb-tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(radio-usb-tests PRIVATE LAB_ENABLE_BLADERF_RF=1)
target_compile_options(radio-usb-tests PRIVATE -Wall -Wextra -Werror)
target_link_libraries(radio-usb-tests PRIVATE GTest::gtest_main PkgConfig::BLADERF Threads::Threads)
foreach(fn set_usb_reset_on_open open close get_fpga_size get_board_name device_speed fw_version fpga_version
  get_loopback enable_module set_sample_rate set_frequency set_gain set_bandwidth sync_config get_timestamp sync_tx sync_rx)
  target_link_options(radio-usb-tests PRIVATE "-Wl,--wrap=bladerf_${fn}")
endforeach()
add_test(NAME radio-usb-contract COMMAND radio-usb-tests)
set_tests_properties(radio-usb-contract PROPERTIES TIMEOUT 20)
